#include <iostream>
using namespace std;

void selectionSort(int arr[], int n)
{
    for(int i=0;i<n-1;i++)
    {
        int min=i;

        for(int j=i+1;j<n;j++)
        {
            if(arr[j]<arr[min])
                min=j;
        }

        swap(arr[i],arr[min]);
    }
}

int main()
{
    int n;

    cout<<"Enter number of elements: ";
    cin>>n;

    int arr[n];

    cout<<"Enter elements:\n";
    for(int i=0;i<n;i++)
        cin>>arr[i];

    selectionSort(arr,n);

    cout<<"\nSorted Array:\n";
    for(int i=0;i<n;i++)
        cout<<arr[i]<<" ";

    cout<<"\n\nTime Complexity:";
    cout<<"\nBest Case    : O(n^2)";
    cout<<"\nAverage Case : O(n^2)";
    cout<<"\nWorst Case   : O(n^2)";

    return 0;
}
